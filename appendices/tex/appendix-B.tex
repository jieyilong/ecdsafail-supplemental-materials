%% Standalone build of Appendix B of the ECDSA.Fail paper.
%%
%% Build (three passes; bibtex is needed for the citations):
%%     pdflatex appendix-B && bibtex appendix-B \
%%       && pdflatex appendix-B && pdflatex appendix-B
%%
%% Requires the LNCS package for llncs.cls and splncs04.bst; on TeX Live
%% install it with:  tlmgr install llncs   (or tlmgr --usermode install llncs)
%%
%% Produces appendix-B.pdf, published in this repository as
%%     ../anatomy_of_agent_identified_quantum_circuit_optimizations.pdf
%%
%% The section body lives in appendix-B-content.tex and the main paper's
%% numbering in main-refs.tex. Upstream both are generated from the paper
%% source by the paper repository's tools/sync-appendix-pdfs.py; the copies
%% here are a snapshot of that output, so edits made in this repository do
%% not flow back into the paper.

%% Numbering of the main paper, so references into it render as real numbers
%% (\newlabel is a preamble-only command, hence its placement here).
